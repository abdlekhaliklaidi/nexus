import { Component, Input, Output, EventEmitter } from '@angular/core';
import { CommonModule } from '@angular/common';
import { Router } from '@angular/router';

export interface Product {
  id: string;
  name: string;
  description: string;
  price: number;
  quantity: number;
  userId: string;
  imageUrls: string[];
}

@Component({
  selector: 'app-product-card',
  standalone: true,
  imports: [CommonModule],
  template: `
    <div 
      (click)="navigateToDetails()"
      class="group relative w-60 bg-white rounded-2xl overflow-hidden border border-slate-100 shadow-sm hover:shadow-xl transition-all duration-300 hover:-translate-y-1.5 cursor-pointer flex flex-col">
      
      <!-- Image Container -->
      <div class="relative w-full h-44 bg-slate-100 overflow-hidden">
        <img 
          [src]="primaryImage" 
          [alt]="product.name" 
          class="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500 ease-out" 
          loading="lazy" />
        
        <span class="absolute bottom-2.5 left-2.5 bg-slate-900/80 backdrop-blur-md text-white text-xs font-bold px-2.5 py-1 rounded-full">
          \${{ product.price | number:'1.2-2' }}
        </span>

        <!-- Action Icons (Owner/Seller Only) -->
        <div *ngIf="isOwner" class="absolute top-2.5 right-2.5 flex gap-1.5 opacity-0 group-hover:opacity-100 -translate-y-1 group-hover:translate-y-0 transition-all duration-200">
          <button 
            (click)="onEdit($event)" 
            title="Edit Product"
            class="w-8 h-8 rounded-full bg-white/90 backdrop-blur-sm text-blue-600 hover:bg-blue-600 hover:text-white flex items-center justify-center shadow-md transition-all duration-200 hover:scale-110">
            <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M15.232 5.232l3.536 3.536m-2.036-5.036a2.5 2.5 0 113.536 3.536L6.5 21.036H3v-3.572L16.732 3.732z"/>
            </svg>
          </button>
          
          <button 
            (click)="onDelete($event)" 
            title="Delete Product"
            class="w-8 h-8 rounded-full bg-white/90 backdrop-blur-sm text-red-600 hover:bg-red-600 hover:text-white flex items-center justify-center shadow-md transition-all duration-200 hover:scale-110">
            <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"/>
            </svg>
          </button>
        </div>
      </div>

      <!-- Card Body -->
      <div class="p-3.5 flex flex-col gap-1.5 flex-grow">
        <h3 class="text-sm font-semibold text-slate-800 truncate" [title]="product.name">
          {{ product.name }}
        </h3>
        
        <p class="text-xs text-slate-500 line-clamp-1">
          {{ truncatedDescription }}
        </p>

        <div class="mt-auto pt-2 flex justify-between items-center text-xs">
          <span 
            class="font-semibold" 
            [ngClass]="product.quantity > 0 ? 'text-emerald-600' : 'text-red-500'">
            {{ product.quantity > 0 ? (product.quantity + ' in stock') : 'Out of stock' }}
          </span>
          <span class="text-indigo-600 font-semibold group-hover:translate-x-0.5 transition-transform">
            View &rarr;
          </span>
        </div>
      </div>
    </div>
  `
})
export class ProductCardComponent {
  @Input({ required: true }) product!: Product;
  @Input() currentUserId: string | null = null;
  @Input() currentUserRole: string | null = null;

  @Output() edit = new EventEmitter<Product>();
  @Output() delete = new EventEmitter<string>();

  constructor(private router: Router) {}

  get isOwner(): boolean {
    return this.currentUserRole === 'SELLER' && this.product?.userId === this.currentUserId;
  }

  get truncatedDescription(): string {
    if (!this.product?.description) return '';
    return this.product.description.length > 20 
      ? this.product.description.substring(0, 20) + '...' 
      : this.product.description;
  }

  get primaryImage(): string {
    return (this.product?.imageUrls && this.product.imageUrls.length > 0)
      ? this.product.imageUrls[0]
      : 'assets/placeholder-product.png';
  }

  navigateToDetails(): void {
    this.router.navigate(['/products', this.product.id]);
  }

  onEdit(event: Event): void {
    event.stopPropagation();
    this.edit.emit(this.product);
  }

  onDelete(event: Event): void {
    event.stopPropagation();
    if (confirm(`Delete "${this.product.name}"?`)) {
      this.delete.emit(this.product.id);
    }
  }
}
